get_filename_component(tls_workspace "$ENV{ZEPHYR_BASE}" DIRECTORY)
set(APOLLO_TLS36_TEST_SOURCE "${tls_workspace}/modules/crypto/mbedtls-3.6" CACHE PATH "Mbed TLS 3.6.6 host-test sources")
if(NOT EXISTS "${APOLLO_TLS36_TEST_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Set APOLLO_TLS36_TEST_SOURCE to workspace Mbed TLS 3.6.6")
endif()
set(ENABLE_PROGRAMS OFF)
set(ENABLE_TESTING OFF)
set(MBEDTLS_AS_SUBPROJECT ON)
set(MBEDTLS_TARGET_PREFIX host36_)
set(MBEDTLS_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/mbedtls/apollo_mbedtls_config.h" CACHE FILEPATH "Apollo TLS config" FORCE)
add_subdirectory("${APOLLO_TLS36_TEST_SOURCE}" library)
foreach(target IN ITEMS host36_mbedcrypto host36_mbedx509 host36_mbedtls)
    target_include_directories(${target} PUBLIC ../../third_party/mbedtls)
endforeach()
add_executable(tls36_tests tls36_tests.c ../../backends/support/zephyr/tls_stack.c)
target_include_directories(tls36_tests PRIVATE shim ../..)
target_link_libraries(tls36_tests PRIVATE host36_mbedtls)
target_compile_options(tls36_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME mbedtls36_real_library_and_platform COMMAND tls36_tests)
